import requests
from openpyxl import Workbook
from openpyxl.styles import PatternFill, Alignment, Font

# ===== API =====
url = "http://172.31.91.110/OP3_MMA/api/yield/yy.php"
res = requests.get(url)
data = res.json()

product = data["product"]
weeks = data["weeks"]

# ===== Excel =====
wb = Workbook()
ws = wb.active
ws.title = "Yield Report"

# ===== Style =====
yellow_fill = PatternFill(start_color="FFFF00", end_color="FFFF00", fill_type="solid")
gray_fill = PatternFill(start_color="D9D9D9", end_color="D9D9D9", fill_type="solid")
red_fill = PatternFill(start_color="F4CCCC", end_color="F4CCCC", fill_type="solid")

center = Alignment(horizontal="center", vertical="center")
bold = Font(bold=True)

# ===== Header =====
ws["B2"] = "WW"
ws["B2"].font = bold

start_col = 5  # เริ่มที่ column E

for i, w in enumerate(weeks):
    col = start_col + i
    ws.cell(row=2, column=col, value=w["ww"]).fill = yellow_fill
    ws.cell(row=2, column=col).alignment = center

# ===== Static Labels =====
labels = ["IN", "OUT", "YIELD", "YIELD Goal"]
for i, label in enumerate(labels):
    ws.cell(row=3 + i, column=2, value=label).font = bold

# ===== Failure Labels =====
fail_labels = ["RF Match", "Calibrate", "High Bandwidth", "TDR", "Noise", "SUM"]
for i, label in enumerate(fail_labels):
    ws.cell(row=8 + i, column=2, value=label)

# ===== Fill Data =====
for i, w in enumerate(weeks):
    col = start_col + i

    # IN / OUT
    ws.cell(row=3, column=col, value=w["in"])
    ws.cell(row=4, column=col, value=w["out"])

    # YIELD
    yield_val = w["yield"]
    if yield_val is not None:
        ws.cell(row=5, column=col, value=f"{yield_val}%")
    else:
        ws.cell(row=5, column=col, value="")

    # Goal
    ws.cell(row=6, column=col, value=f"{w['yield_goal']}%")

    # Failure
    fail = w["failure"]
    ws.cell(row=8, column=col, value=fail["RF Match"])
    ws.cell(row=9, column=col, value=fail["Calibrate"])
    ws.cell(row=10, column=col, value=fail["High Bandwidth"])
    ws.cell(row=11, column=col, value=fail["TDR"])
    ws.cell(row=12, column=col, value=fail["Noise"])
    ws.cell(row=13, column=col, value=fail["sum"])

# ===== Product Name =====
ws["A5"] = product
ws["A5"].fill = red_fill

# ===== Adjust Column Width =====
for col in range(1, 50):
    ws.column_dimensions[chr(64 + col)].width = 12

# ===== Save =====
wb.save("yield_report.xlsx")

print("✅ Export Excel เรียบร้อย: yield_report.xlsx")
